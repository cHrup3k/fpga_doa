cd ~/fpga/2mic

rm -rf build

python3 pdm_udp_stereo_shared.py \
    --build \
    --pdm-clk-pins "F5 E6 H16 F18 B3 K4 F1 E4" \
    --pdm-data-pins "E18 G16 E5 F4 K5 A2 F2 E1" \
    --eth-phy 0 \
    --uart-name=crossover \
    --with-uartbone \
    --uart-baudrate=115200 \
    --integrated-rom-size=0xc000 \
    --l2-size=2048 \
    --csr-csv=csr.csv
